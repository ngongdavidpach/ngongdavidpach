Instagram and Facebook sellers in Juba can book JunubLogistics deliveries straight from their Shopify, DM workflow, or custom storefront with a single JSON POST.

Content-Type: application/json

  "recipient_name": "Achol Deng",
  "recipient_phone": "+211922123456",
  "pickup_lat": 4.851, "pickup_lng": 31.582,
  "pickup_label": "Jumia Juba Hub, Juba Market",
  "dropoff_lat": 4.844, "dropoff_lng": 31.539,
  "dropoff_label": "Gudele Block 4",

  "package_size": "medium",
  "description": "Jumia order #SS2391 — shoes"
}
